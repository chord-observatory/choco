{# Shared macros for the services strip, the landing table and the /service/ pages. Colour is a *tone*, never a hex value: ok / bad / warn / info / maint / off. The *_tone macros map each subsystem's health strings onto those six, tag() turns a tone into a `tag-` class, and static/choco.css paints it for both themes. #} {# monitor_* cover the hardware monitors (FPGA, PDB) and the DATA scan: ok/down/unconfigured/unknown plus per-monitor degraded states (no_timing, no_states) that all render as warn. #} {% macro monitor_tone(h) -%} {%- if h == 'ok' -%}ok {%- elif h == 'down' -%}bad {%- elif h in ('unconfigured', 'unknown') -%}off {%- else -%}warn {%- endif -%} {%- endmacro %} {% macro monitor_label(h) -%} {%- if h == 'ok' -%}up {%- elif h == 'unconfigured' -%}not configured {%- else -%}{{ h | replace('_', ' ') }} {%- endif -%} {%- endmacro %} {# job states: ok; degraded (dependency/input trouble, exit 2) and stale warn; failed (config/bug) bad; never_run/unknown off -- unless a run is in flight with no verdict yet, which is info ("running", see job_label). Jobs say "ok"/"failed", not "up"/"down" -- up/down is for the daemon monitors. #} {% macro job_tone(h, running=false) -%} {%- if running and h in ('unknown', 'never_run') -%}info {%- elif h == 'ok' -%}ok {%- elif h in ('degraded', 'stale') -%}warn {%- elif h in ('never_run', 'unknown') -%}off {%- else -%}bad {%- endif -%} {%- endmacro %} {# bffs source statuses (from the job's run file): ok; degraded (ran, abstained or lost coverage) and skipped (not run: no N² file) warn; unknown off; anything else bad. #} {% macro source_tone(h) -%} {%- if h == 'ok' -%}ok {%- elif h in ('degraded', 'skipped') -%}warn {%- elif h == 'unknown' -%}off {%- else -%}bad {%- endif -%} {%- endmacro %} {# node status (state.NodeStatus values). Idle is a chosen state, not a warning, so it is neutral like unknown; the strip's NODES roll-up (web._nodes_health) still reports an all-idle cluster as warn. #} {% macro node_tone(status) -%} {%- if status == 'started' -%}ok {%- elif status == 'down' -%}bad {%- elif status == 'syncing' -%}info {%- else -%}off {%- endif -%} {%- endmacro %} {# What the operator asked for when it differs from what is true: the Start switch says the wish, the status tag the fact, and when they disagree nothing else points at the gap. Empty while syncing (the gap is being closed) and while unknown (nothing has been polled). #} {% macro node_wants(node) -%} {%- set st = node.status.value -%} {%- if node.started and st in ('idle', 'down') -%}wants started {%- elif not node.started and st == 'started' -%}wants idle {%- endif -%} {%- endmacro %} {# running: a run is in flight; the health shown is the last completed one, so only a badge with no verdict yet says "running". #} {% macro job_label(h, running=false) -%} {%- if running and h in ('unknown', 'never_run') -%}running {%- elif h == 'never_run' -%}not run {%- elif h -%}{{ h }} {%- else -%}unknown {%- endif -%} {%- endmacro %} {% macro since(ts, now) -%} {%- if ts -%} {%- set delta = (now - ts) | int -%} {%- if delta < 60 -%}{{ delta }}s ago {%- elif delta < 3600 -%}{{ (delta / 60) | int }}m ago {%- elif delta < 86400 -%}{{ (delta / 3600) | int }}h ago {%- else -%}{{ (delta / 86400) | int }}d ago {%- endif -%} {%- else -%}never {%- endif -%} {%- endmacro %} {# The status tag. `word` is the state ("all up", "degraded", "started"); `label` names the service in the strip ("NODES"). An with href, else a . quiet: an ok / info / off tag drops the word and its tint and keeps the label and a dot in its tone -- the word stays in the tooltip and in aria-label -- so the strip is quiet unless something is warn or bad, and nine services fit one nav line. extra: further classes (the node badges carry a `status-` hook). #} {% macro tag(tone, word, label=none, href=none, title=none, quiet=false, extra='') -%} {%- set hush = quiet and tone in ('ok', 'info', 'off') -%} {%- set el = 'a' if href else 'span' -%} <{{ el }}{% if href %} href="{{ href }}"{% endif %} class="tag tag-{{ tone }}{% if hush %} quiet{% endif %}{% if extra %} {{ extra }}{% endif %}"{% if label %} aria-label="{{ label }}: {{ word }}"{% endif %}{% if title %} title="{{ title }}"{% endif %}>{% if label %}{{ label }}{% endif %}{% if not hush %}{{ word }}{% endif %} {%- endmacro %}
